INTRODUCCIÓN

Se implementó dos variantes de una Queue concurrente en Rust: una versión bloqueante utilizando Mutex y CondVar y una
versión no bloqueante usando CAS. Se realizaron los siguientes tests para comparar el desempeño de ambas
implementaciones:

Los tests fueron ejecutados con el siguiente código de bash:

#!/bin/bash

# ============================================
# CONFIG
# ============================================

QUEUE_TYPES=("blocking" "nonblocking")

TESTS=(
  "4 4 100"
  "4 4 100000"
  "4 4 1000000"

  "1 1 100000"
  "2 2 100000"

  "4 1 100000"

  "8 8 100000"
  "8 2 100000"
  "16 2 100000"

  "2 8 100000"
  "2 16 100000"
)

# ============================================
# RUN
# ============================================

echo "Building release binary..."
cargo build --release

echo ""
echo "========================================"
echo "RUNNING BENCHMARKS"
echo "========================================"

for queue in "${QUEUE_TYPES[@]}"
do
  echo ""
  echo "########################################"
  echo "QUEUE TYPE: $queue"
  echo "########################################"

  for test in "${TESTS[@]}"
  do
    read producers consumers items <<< "$test"

    echo ""
    echo "----------------------------------------"
    echo "P=$producers C=$consumers ITEMS=$items"
    echo "----------------------------------------"

    ./target/release/tp5 \
      --queue "$queue" \
      --producers "$producers" \
      --consumers "$consumers" \
      --items "$items"
  done
done
